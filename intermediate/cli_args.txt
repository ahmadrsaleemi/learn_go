package main

import (
	"flag"
	"fmt"
	"os"
)

func main() {
	fmt.Println("Command: ", os.Args[0])

	arg_num := 1
	for _, arg := range os.Args{
		fmt.Println("arg # : ", arg_num, " ", arg)
		arg_num += 1
	}

	//define flags
	var name string
	var age int

	flag.StringVar(&name, "name", "John", "Name of the user")
	flag.IntVar(&age, "age", 32, "Age of the user")

	flag.Parse()

	fmt.Println("Name:",name)
	fmt.Println("Age:",age)
}
