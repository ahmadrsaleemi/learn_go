package main

import (
	"crypto/rand"
	"crypto/sha256"
	"crypto/sha512"
	"encoding/base64"
	"fmt"
	"io"
)

func main() {
	password := "testing123"
	hashedPassword256 := sha256.Sum256([]byte(password))
	hashedPassword512 := sha512.Sum512([]byte(password))

	fmt.Println(password)
	fmt.Println(hashedPassword256)
	fmt.Printf("SHA-256 hash hex value: %x\n", hashedPassword256)
	fmt.Printf("SHA-512 hash hex value: %x\n", hashedPassword512)

	fmt.Println("Customized password!")

	salt, err := generateSalt()

	if err != nil {
		fmt.Println("Error generating salt: ", err)
		return
	}

	hash := hashPasword(password, salt)
	saltStr := base64.StdEncoding.EncodeToString(salt)
	fmt.Println("Salt: ", saltStr)
	fmt.Println("Hash: ", hash)
}

func generateSalt() ([]byte, error) {
	salt := make([]byte, 16)
	_, err := io.ReadFull(rand.Reader, salt)

	if err != nil {
		return nil, err
	}
	return salt, nil
}

func hashPasword(password string, salt []byte) string {
	saltedPassword := append(salt, []byte(password)...)
	hash := sha256.Sum256(saltedPassword)
	return base64.StdEncoding.EncodeToString(hash[:])
}
