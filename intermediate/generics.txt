package main

import "fmt"

func swap[T any](a, b T) (T , T) {
	return b, a
}

type Stack[T any] struct {
	elements []T
}

func (s *Stack[T]) push(element T) {
	s.elements = append(s.elements, element)
}

func (s *Stack[T]) pop() (T, bool) {
	if len(s.elements) == 0 {
		var zero T
		return zero, false
	}
	element := s.elements[len(s.elements) - 1] //get last element from stack
	s.elements = s.elements[: len(s.elements) - 1]
	return element, true
}

func (s *Stack[T]) isEmpty() bool {
	return len(s.elements) == 0
}

func (s Stack[T]) printStack() {
	if len(s.elements) == 0 {
		fmt.Println("The stack is empty")
		return
	}
	fmt.Println("tokens behind:")
	for _, element := range s.elements {
		fmt.Println(element)
	}
}

func main() {
	fmt.Println("Experimenting with Generics in Go!")

	x := 1
	y := 2
	x, y = swap(x, y)

	fmt.Println(x , "  " , y)

	name1, name2 := "John", "Doe"
	fmt.Println("Names: " , name1, " " , name2)
	name1, name2 = swap(name1, name2)
	fmt.Println("Names: " , name1, " " , name2)

	token := Stack[int]{}
	token.push(10)
	token.push(14)
	token.push(19)
	token.push(28)
	token.printStack()

	tokenNumber, stackStatus := token.pop()
	fmt.Println("One token popped")
	if stackStatus {
		fmt.Println("Token number: ", tokenNumber , "  come to counter")
	}
	token.printStack()
	tokenNumber, stackStatus = token.pop()
	fmt.Println("One token popped")
	if stackStatus {
		fmt.Println("Token number: ", tokenNumber , "  come to counter")
	}
	token.printStack()


	names := Stack[string]{}
	names.push("Jens")
	names.push("Johansson")
	names.push("Linda")
	names.push("Karl")

	names.printStack()

}
