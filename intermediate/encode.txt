package main

import (
	"encoding/base64"
	"fmt"
)

func main() {
	data := []byte("hello, experimenting with Go!")

	encoded := base64.StdEncoding.EncodeToString(data)

	fmt.Println(encoded)

	decoded, err := base64.StdEncoding.DecodeString(encoded)

	if err != nil {
		fmt.Println("Error found: ", err)
	}

	fmt.Println(string(decoded))
}
