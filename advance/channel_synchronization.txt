func main (){
	data := make(chan string)

	go func(){
		for i:= range 5 {
			data <- "Stock ID: " + string('0' + i)
			time.Sleep(100 * time.Millisecond)
		}
		close(data)
	}()

	time.Sleep(5 * time.Second)


	for value := range data {
		fmt.Println("Received... ", value)
	}
}
